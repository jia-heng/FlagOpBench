# 性能基线平台设计文档

## 1. 目标

在多硬件平台上（NVIDIA、昇腾、沐曦等），对 55 个关键算子建立统一性能基线。通过统一的测试用例和框架，驱动各平台的算子实现，输出可横向对比的精度与性能数据。

### 1.1 设计原则

- **通用性优先**：框架与用例设计平台无关，差异收敛在后端适配层
- **torch 优先**：算子调用优先使用各厂商定制的 PyTorch 接口，torch 中无覆盖的再走 vLLM 或第三方
- **双轨计时**：device time（主指标）+ wall clock（含 launch overhead，辅助参考）
- **用例统一**：同一组 case 驱动不同后端，结果横向可比
- **精度与性能并重**：每次测试同时输出正确性校验和性能数据

### 1.2 算子分层与开发优先级

| 层级 | 调用来源 | 示例 | 优先级 |
|------|----------|------|--------|
| 基础算子 | 各平台 torch | mm, bmm, layernorm, softmax, gelu | P0 - 首批开发 |
| 大模型算子 | 各平台定制 vLLM | flash_attention, fused_moe, paged_kv_cache | P1 - 后续补充 |
| 第三方算子 | 独立仓库 | FlashMLA, DeepGEMM, flash-linear-attention | P2 - 按需接入 |

## 3. 公共框架设计

### 3.1 双轨计时器 (framework/timer.py)

```python
import time
import torch

class BenchmarkTimer:
    """双轨计时：device event time + wall clock time"""

    def __init__(self, device, warmup=10, iters=100):
        self.device = device
        self.warmup = warmup
        self.iters = iters

    def measure(self, fn):
        # Warmup
        for _ in range(self.warmup):
            fn()
        self._sync()

        device_times = self._measure_device(fn)
        wall_times = self._measure_wall(fn)

        return {
            "device_time": self._compute_stats(device_times),
            "wall_time": self._compute_stats(wall_times),
        }

    def _measure_device(self, fn):
        """Device event 计时，不含 launch overhead"""
        times = []
        for _ in range(self.iters):
            start_event = self._create_event()
            end_event = self._create_event()
            self._record_event(start_event)
            fn()
            self._record_event(end_event)
            self._sync()
            times.append(self._elapsed(start_event, end_event))
        return times

    def _measure_wall(self, fn):
        """Wall clock 计时，含 launch overhead"""
        times = []
        for _ in range(self.iters):
            self._sync()
            t0 = time.perf_counter()
            fn()
            self._sync()
            t1 = time.perf_counter()
            times.append((t1 - t0) * 1000)  # ms
        return times

    def _sync(self):
        """平台相关的同步操作，由子类或后端适配"""
        torch.cuda.synchronize()  # 默认 CUDA，可由后端覆盖

    def _create_event(self):
        return torch.cuda.Event(enable_timing=True)

    def _record_event(self, event):
        event.record()

    def _elapsed(self, start, end):
        return start.elapsed_time(end)  # ms

    @staticmethod
    def _compute_stats(times):
        times_sorted = sorted(times)
        n = len(times_sorted)
        return {
            "mean_ms": sum(times_sorted) / n,
            "median_ms": times_sorted[n // 2],
            "min_ms": times_sorted[0],
            "max_ms": times_sorted[-1],
            "p99_ms": times_sorted[int(n * 0.99)],
        }
```

> 设计要点：timer 的 `_sync`、`_create_event`、`_record_event`、`_elapsed` 方法由后端适配层覆盖，框架本身不绑定 CUDA。

### 3.2 精度校验 (framework/validator.py)

```python
class Validator:
    """精度校验：对比 golden reference"""

    def __init__(self, golden_backend="cpu", tolerance=None):
        """
        golden_backend: "cpu" 使用 CPU fp32 计算作为标杆
                        "nvidia" 使用 NV 结果作为标杆
        tolerance: 自定义容差，None 则根据 dtype 自动选取
        """
        self.golden_backend = golden_backend
        self.tolerance = tolerance

    def validate(self, operator, inputs, actual_output):
        """
        返回:
            passed: bool
            max_abs_error: float
            max_rel_error: float
            cosine_similarity: float
        """
        golden = self._compute_golden(operator, inputs)
        return self._compare(golden, actual_output)

    def _default_tolerance(self, dtype):
        return {
            "fp32": 1e-5,
            "fp16": 1e-3,
            "bf16": 1e-2,
            "fp8": 1e-1,
        }.get(dtype, 1e-2)
```

```python
        """返回 torch device"""
        ...

    @abstractmethod

        ...
```

### 3.4 测试执行引擎 (framework/runner.py)

```python
class BenchmarkRunner:
    """测试执行引擎：加载 case → 调用后端 → 计时 → 校验 → 输出"""

    def __init__(self, backend: Backend, validator: Validator):
        self.backend = backend
        self.validator = validator

    def run_case(self, case_config: dict):
        """执行单个 case"""
        op_name = case_config["operator"]
        op_fn = self.backend.get_operator(op_name, case_config["level"])
        timer = self.backend.create_timer(
            warmup=case_config.get("warmup", 10),
            iters=case_config.get("iters", 100),
        )

        results = []
        for scenario in case_config["scenarios"]:
            inputs = self._prepare_inputs(scenario)
            # 精度校验
            output = op_fn(**inputs)
            accuracy = self.validator.validate(op_name, inputs, output)
            # 性能测试
            fn = lambda: op_fn(**inputs)
            perf = timer.measure(fn)

            results.append({
                "scenario": scenario["name"],
                "params": scenario,
                "accuracy": accuracy,
                "performance": perf,
            })
        return results
```

**效率计算逻辑 (framework/roofline.py)：**

```python
class RooflineAnalyzer:
    """基于 Roofline 模型计算算子效率"""

    def __init__(self, hw_specs: dict):
        self.specs = hw_specs

    def analyze(self, op_meta: dict, device_time_ms: float) -> dict:
        """
        op_meta 包含:
          - flops: 该算子此 shape 下的理论计算量 (FLOPs)
          - bytes: 该算子此 shape 下的访存量 (Bytes)
          - dtype: 数据类型
        返回:
          - achieved_tflops: 实际算力
          - achieved_bandwidth_tb_s: 实际带宽
          - compute_efficiency_pct: 算力效率 (%)
          - memory_efficiency_pct: 带宽效率 (%)
          - bound: "compute" | "memory" (瓶颈判断)
        """
        seconds = device_time_ms / 1000.0

        achieved_tflops = op_meta["flops"] / seconds / 1e12
        achieved_bw = op_meta["bytes"] / seconds / 1e12  # TB/s

        peak_tflops = self.specs["compute"][f"{op_meta['dtype']}_tflops"]
        peak_bw = self.specs["memory"]["hbm_bandwidth_tb_s"]

        compute_eff = achieved_tflops / peak_tflops * 100
        memory_eff = achieved_bw / peak_bw * 100

        # 算术强度判断瓶颈
        arithmetic_intensity = op_meta["flops"] / op_meta["bytes"]  # FLOPs/Byte
        ridge_point = peak_tflops * 1e12 / (peak_bw * 1e12)  # FLOPs/Byte

        bound = "compute" if arithmetic_intensity > ridge_point else "memory"

        return {
            "achieved_tflops": round(achieved_tflops, 2),
            "achieved_bandwidth_tb_s": round(achieved_bw, 3),
            "compute_efficiency_pct": round(compute_eff, 1),
            "memory_efficiency_pct": round(memory_eff, 1),
            "arithmetic_intensity": round(arithmetic_intensity, 1),
            "bound": bound,
        }
```

> 每个算子需在注册时提供 `compute_flops(params)` 和 `compute_bytes(params)` 方法，用于根据 shape 自动计算理论计算量和访存量。

## 7. 跨平台适配要点

| 适配项 | 框架层处理 | 后端层处理 |
|--------|-----------|-----------|
| 计时方式 | timer 抽象接口 | 各平台 event API 实现 |

### 8.2 回归检测

```python
class RegressionDetector:
    """对比两次测试结果，检测性能回归"""

    def __init__(self, threshold_pct=5.0):
        """
        threshold_pct: 性能劣化超过此百分比则标记为回归
        """
        self.threshold_pct = threshold_pct

    def compare(self, baseline_result: dict, current_result: dict) -> list:
        """
        返回回归项列表:
          - operator, scenario
          - baseline_ms, current_ms
          - regression_pct
          - status: "regression" | "improved" | "stable"
        """
        regressions = []
        for b, c in self._match_scenarios(baseline_result, current_result):
            b_time = b["performance"]["device_time"]["mean_ms"]
            c_time = c["performance"]["device_time"]["mean_ms"]
            delta_pct = (c_time - b_time) / b_time * 100

            status = "stable"
            if delta_pct > self.threshold_pct:
                status = "regression"
            elif delta_pct < -self.threshold_pct:
                status = "improved"

            regressions.append({
                "operator": b["operator"],
                "scenario": b["scenario"],
                "baseline_ms": b_time,
                "current_ms": c_time,
                "delta_pct": round(delta_pct, 2),
                "status": status,
            })
        return regressions
```

