#### 1.1 flashattention (基础版)
```python
# 算子: flashattention
# 来源: flash-attn 官方库
# 实现: flash_attn.flash_attn_func

from flash_attn import flash_attn_func

def forward(q, k, v, causal=True):
    """
    q, k, v: (batch, seq_len, num_heads, head_dim)
    PyTorch 官方: F.scaled_dot_product_attention (PyTorch 2.0+)
    优先级: PyTorch > flash-attn > 手动实现
    """
    if hasattr(F, 'scaled_dot_product_attention'):
        # PyTorch 2.0+ 官方优化实现
        return F.scaled_dot_product_attention(q, k, v, is_causal=causal)
    elif HAS_FLASH_ATTN:
        return flash_attn_func(q, k, v, causal=causal)
    else:
        # Fallback: 手动实现
        scores = torch.matmul(q, k.transpose(-2, -1)) / math.sqrt(head_dim)
        if causal:
            mask = torch.triu(torch.ones(...), diagonal=1).bool()
            scores.masked_fill_(mask, float('-inf'))
        attn = F.softmax(scores, dim=-1)
        return torch.matmul(attn, v)
```

**实施要点**:
- ✅ 优先使用 PyTorch 2.0+ 的 `F.scaled_dot_product_attention`
- ✅ 次选 flash-attn 库（需要单独安装）
- ✅ Fallback 到手动实现

**测试用例**:
- Llama-3.1: (batch=1-128, seq=1-8192, heads=32, dim=128)
- DeepSeek-V3: (batch=1-64, seq=1-4096, heads=128, dim=128)

### 2. MoE 融合算子 (3 个)

#### 2.1 fused_moe
```python
# 算子: fused_moe
# 来源: vLLM cutlass_moe
# 实现: torch.ops._C.cutlass_moe_mm (vLLM) 或手动实现

def forward(x, expert_weights, router_logits, top_k=2):
    """
    融合 MoE: routing + expert computation
    优先级: 手动实现 (组合已有算子)
    
    步骤:
    1. TopK routing
    2. Grouped matmul per expert
    3. Weighted sum
    """
    # 1. TopK routing
    topk_weights, topk_ids = torch.topk(router_logits, top_k, dim=-1)
    topk_weights = F.softmax(topk_weights, dim=-1)
    
    # 2. Expert computation (可复用 grouped_matmul)
    expert_outputs = grouped_matmul(x, expert_weights, topk_ids)
    
    # 3. Weighted sum (复用 moe_sum)
    output = moe_sum(expert_outputs, topk_weights)
    
    return output
```

**实施要点**:
- ✅ 组合已实现的算子: topk + grouped_matmul + moe_sum
- ✅ 纯 PyTorch 实现，易于移植
- 📋 后续可对接 vLLM CUDA kernel 优化

#### 2.2 MoeAlignBlockSize
```python
# 算子: MoeAlignBlockSize
# 来源: vLLM
# 实现: torch.ops._C.moe_align_block_size

def forward(topk_ids, block_size=64):
    """
    对齐 MoE block size，提高 GPU 利用率
    优先级: 手动实现 (padding)
    """
    # 简化版: padding to block_size
    num_tokens = topk_ids.shape[0]
    padded_size = ((num_tokens + block_size - 1) // block_size) * block_size
    if padded_size > num_tokens:
        padding = torch.zeros(padded_size - num_tokens, dtype=topk_ids.dtype, device=topk_ids.device)
        topk_ids = torch.cat([topk_ids, padding])
    return topk_ids
```

#### 3.3 per_token_group_fp8_quant
```python
# 算子: per_token_group_fp8_quant
# 来源: vLLM dynamic quantization
# 实现: PyTorch 手动实现

def forward(x, group_size=128):
    """
    Per-token FP8 动态量化
    优先级: 手动实现
    """
    # 计算每组的 scale
    x_reshape = x.reshape(-1, group_size)
    scales = x_reshape.abs().max(dim=-1, keepdim=True)[0] / 127.0
    
    # 量化
    x_quant = (x_reshape / scales).round().clamp(-127, 127).to(torch.int8)
    
    return x_quant, scales
```

def forward(x, clamp_min=-10, clamp_max=10):

    SiLU + Mul + Clamp

#### 4.3 kv_rms_norm_rope_cache
```python
# 算子: kv_rms_norm_rope_cache
# 来源: vLLM
# 实现: RMSNorm + RoPE + Cache

def forward(k, v, norm_weight, cos, sin, cache):
    """
    融合 KV RMSNorm + RoPE + Cache
    优先级: 组合已实现算子
    """
    # 1. RMSNorm
    k_norm = F.rms_norm(k, [k.shape[-1]], norm_weight)
    v_norm = F.rms_norm(v, [v.shape[-1]], norm_weight)
    
    # 2. RoPE (复用 rope operator)
    k_rope = apply_rope(k_norm, cos, sin)
    
    # 3. Cache (简单拼接)
    k_cached = torch.cat([cache['k'], k_rope], dim=1)
    v_cached = torch.cat([cache['v'], v_norm], dim=1)
    
    return k_cached, v_cached
```
