# baseline/operators/model/fused_moe.py
"""Fused MoE 算子 - 使用 vLLM CUDA kernel"""

import torch
from baseline.operators.registry import BaseOperator, register_operator

try:
    import torch.ops._C as vllm_ops
    HAS_VLLM_MOE = hasattr(vllm_ops, 'cutlass_moe_mm')
except (ImportError, AttributeError):
    HAS_VLLM_MOE = False
    print("Warning: vLLM CUDA ops not available for fused_moe, using PyTorch fallback")

@register_operator("fused_moe")
class FusedMoEOperator(BaseOperator):
    """Fused MoE GEMM with TopK routing"""

    @property
    def name(self) -> str:
        return "fused_moe"

    def forward(self, x: torch.Tensor, expert_weights: torch.Tensor,
                top_k_indices: torch.Tensor, **kwargs) -> torch.Tensor:
        if HAS_VLLM_MOE:
            # 调用 vLLM 官方 CUDA kernel
            return vllm_ops.cutlass_moe_mm(
                x, expert_weights, top_k_indices, ...
            )
        else:
            # Fallback: PyTorch 实现
            # TODO: 实现基础版本
            pass

    def compute_flops(self, num_tokens: int, hidden_size: int,
                      num_experts: int, k: int = 8, **kwargs) -> int:
        # 每个 token 调用 k 个 expert，每个 expert 是 2 个 GEMM
        return 2 * num_tokens * k * hidden_size * hidden_size

    def compute_bytes(self, num_tokens: int, hidden_size: int,
                      num_experts: int, k: int = 8,
                      dtype: str = "bf16", **kwargs) -> int:
        elem_bytes = self.dtype_bytes(dtype)
        # 简化计算：读 input + 读 weights + 写 output
        read_input = num_tokens * hidden_size * elem_bytes
        read_weights = k * num_experts * hidden_size * hidden_size * elem_bytes
        write_output = num_tokens * hidden_size * elem_bytes
        return read_input + read_weights + write_output

    def prepare_inputs(self, num_tokens: int, hidden_size: int,
                       num_experts: int, k: int = 8,
                       dtype: str = "bf16", **kwargs) -> dict:
        torch_dtype = self.get_dtype(dtype)
        x = torch.randn(num_tokens, hidden_size,
                       device=self.device, dtype=torch_dtype)
        expert_weights = torch.randn(
            num_experts, hidden_size, hidden_size,
            device=self.device, dtype=torch_dtype
        )
        top_k_indices = torch.randint(
            0, num_experts, (num_tokens, k),
            device=self.device, dtype=torch.int32
        )
        return {"x": x, "expert_weights": expert_weights,
                "top_k_indices": top_k_indices, "k": k}
